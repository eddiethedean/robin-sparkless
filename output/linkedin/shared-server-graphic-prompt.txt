from sparkless import SparklessServer

limits = {
    "SPARKLESS_MAX_ACTIVE_OPERATIONS": "4"
}
with SparklessServer(env=limits) as server:
    spark = SparkSession.builder.remote(
        server.spark_connect_url
    ).create()
    try:
        check_orders(spark)
    finally:
        spark.stop()
